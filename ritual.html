#magic-bg { position: fixed; top: 0; left: 0; width: 100vw; height: 100vh; pointer-events: none; /* Klicks gehen durch den Canvas durch */ z-index: 0; /* Hinter deinen Inhalten */ opacity: 0.6; } const canvas = document.getElementById('magic-bg'); const ctx = canvas.getContext('2d'); let particles = []; function resizeCanvas() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; } window.addEventListener('resize', resizeCanvas); resizeCanvas(); class Particle { constructor() { this.reset(); } reset() { this.x = Math.random() * canvas.width; this.y = canvas.height + Math.random() * 100; this.size = Math.random() * 3 + 1; this.speedY = Math.random() * 1.5 + 0.5; this.speedX = (Math.random() - 0.5) * 0.8; this.opacity = Math.random() * 0.8 + 0.2; // Hexengrün-Farbpalette this.color = `hsla(${110 + Math.random() * 40}, 100%, 60%, `; } update() { this.y -= this.speedY; this.x += this.speedX; if (this.y < -10) this.reset(); } draw() { ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctx.fillStyle = this.color + this.opacity + ')'; ctx.shadowBlur = 8; ctx.shadowColor = '#00ff66'; ctx.fill(); } } for (let i = 0; i < 60; i++) { particles.push(new Particle()); } function animate() { ctx.clearRect(0, 0, canvas.width, canvas.height); particles.forEach(p => { p.update(); p.draw(); }); requestAnimationFrame(animate); } animate(); /* Glühendes Pulsieren für grüne Hexenmagie */ .witch-glow { text-shadow: 0 0 5px #2ecc71, 0 0 10px #2ecc71, 0 0 20px #27ae60; animation: pulseGlow 3s infinite alternate ease-in-out; } @keyframes pulseGlow { 0% { text-shadow: 0 0 4px #2ecc71, 0 0 8px #2ecc71; } 50% { text-shadow: 0 0 12px #00ff66, 0 0 25px #00ff66, 0 0 35px #1abc9c; } 100% { text-shadow: 0 0 4px #2ecc71, 0 0 8px #2ecc71; } }
